root RIFF {
  magic: text[4] ascii
  size: switch magic {'RIFX' => u32be, _ => u32le}
      valid max magic == 'RF64' ? 4294967295 : size of space - 8
  form: text[4] ascii
  chunks: sized(min(max(size, 4) - 4 or else remaining, remaining)) switch magic {
    'RIFX' => repeat(until end) ChunkBE
    _ => repeat(until end) Chunk
  }
}

            'TPOS' => same as case 'TIT2'
            'COMM' => Comment {
              encoding: u8be enum TextEncoding {
                0 = latin-1, 1 = utf-16 bom, 2 = utf-16 be, 3 = utf-8
              }
              language: text[3] ascii
              text: switch encoding {
                0 => text[remaining] latin1
                1 => text[remaining] bom(utf16be)
                2 => text[remaining] utf16be
                3 => text[remaining] utf8
                _ => text[remaining] encoding unknown
              }
            }
            _ => bytes[remaining]
          }
        }
      }
    }
    'ID3 ' => same as case 'id3 '
    'data' => Samples {
      samples: switch earlier(body.format) {
        0xfffe => switch earlier(body.extra.sub_format.format) {
          1 => switch earlier(body.bits_per_sample) {
            8 => u8be[remaining / 1]
            16 => i16be[remaining / 2]
            24 => i24be[remaining / 3]
            32 => i32be[remaining / 4]
            _ => bytes[remaining]
          }
          3 => switch earlier(body.bits_per_sample) {
            32 => f32be[remaining / 4]
            64 => f64be[remaining / 8]
            _ => bytes[remaining]
          }
          _ => bytes[remaining]
        }
        _ => switch earlier(body.format) {
          1 => switch earlier(body.bits_per_sample) {
            8 => u8be[remaining / 1]
            16 => i16be[remaining / 2]
            24 => i24be[remaining / 3]
            32 => i32be[remaining / 4]
            _ => bytes[remaining]
          }
          3 => switch earlier(body.bits_per_sample) {
            32 => f32be[remaining / 4]
            64 => f64be[remaining / 8]
            _ => bytes[remaining]
          }
          _ => bytes[remaining]
        }
      }  valid finite
          table columns earlier(body.channels), rate earlier(body.sample_rate), rows sample,
              column word channel, named [left, right], describes
              [earlier(body.format), earlier(body.channels), earlier(body.sample_rate),
              earlier(body.bits_per_sample)]
    }
    _ => bytes[remaining]
  }
  pad: bytes[(((size < 4294967295) * (size + 1) or else earlier(body.data_size) + 1) - 1 -
      (((size < 4294967295) * (size + 1) or else earlier(body.data_size) + 1) - 1) / 2 * 2) *
      (0 < remaining)]
}
