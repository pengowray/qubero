root RIFF {
  magic: text[4] ascii
  size: switch magic {'RIFX' => u32be, _ => u32le}
      valid max magic == 'RF64' ? 4294967295 : size of space - 8
  form: text[4] ascii
  chunks: sized(min(max(size, 4) - 4 or else remaining, remaining)) switch magic {
    'RIFX' => repeat(until end) ChunkBE
    _ => repeat(until end) Chunk
  }
}
