template zip

root ZIP {
  records: repeat(until element.signature is 50 4B 05 06) ZipRecord
      (named by signature, contents body, unit record) {
    signature: u32le enum Signature {
      0x04034b50 = local file, 0x08074b50 = data descriptor,
      0x02014b50 = central directory file, 0x06054b50 = end of central directory,
      0x06064b50 = ZIP64 end, 0x07064b50 = ZIP64 locator
    }
    body: switch signature {
      0x04034b50 => LocalFile {
        version_needed: u16le
        flags: u16le flags GeneralPurposeFlags {
          bit 0 = encrypted, bit 3 = data descriptor follows, bit 6 = strong encryption,
          bit 11 = UTF-8 names, bit 13 = directory encryption
        }
        compression: u16le enum CompressionMethod {
          0 = stored, 1 = shrunk, 6 = imploded, 8 = deflate, 9 = deflate64, 12 = bzip2,
          14 = lzma, 93 = zstandard, 95 = xz, 98 = ppmd, 99 = aes
        }
        modified_time: u16le  time dos halves(date modified_date, time modified_time) local
        modified_date: u16le  time dos halves(date modified_date, time modified_time) local
        crc32: u32le
            check crc32 over unpacked data of unpacked_size bytes when
                (1 - (flags / 1 - flags / 2 * 2)) * (1 - (flags / 8 - flags / 16 * 2))
        compressed_size: u32le
        uncompressed_size: u32le
        name_length: u16le
        extra_length: u16le
        name: text[name_length] encoding unknown
        extra: sized(extra_length) repeat(until end) ExtraField (named by id, contents data) {
          id: u16le enum ExtraFieldId {
            0x01 = ZIP64 extended info, 0x07 = AV info, 0x08 = language, 0x09 = OS/2,
            0x0a = NTFS times, 0x0c = OpenVMS, 0x0d = UNIX, 0x0e = stream and fork descriptor,
            0x0f = patch descriptor, 0x14 = PKCS#7 certificates, 0x15 = file signature,
            0x16 = directory signature, 0x17 = strong encryption, 0x19 = certificate list,
            0x65 = IBM attributes, 0x66 = IBM compressed attributes, 0x4690 = POSZIP,
            0x5455 = extended timestamp, 0x5855 = Info-ZIP UNIX, 0x6375 = Unicode comment,
            0x7075 = Unicode name, 0x7855 = Info-ZIP UNIX 2, 0x7875 = Info-ZIP UNIX 3,
            0x9901 = AES encryption, 0xa11e = alignment padding, 0xa220 = growth hint
          }
          size: u16le
          data: sized(size) switch id {
            1 => switch 15 < remaining {
              1 => Zip64Sizes {
                uncompressed_size: u64le
                compressed_size: u64le
                rest: bytes[remaining]
              }
              _ => bytes[remaining]
            }
            _ => bytes[remaining]
          }
        }
        data_size: computed compressed_size * (1 - (4294967294 < compressed_size)) or else
            (extra[id = 1].data.compressed_size or else
            (1 - (flags / 8 - flags / 16 * 2) or else find(50 4B 07 08) + 1) - 1)
        unpacked_size: computed uncompressed_size * (1 - (4294967294 < uncompressed_size)) or
            else extra[id = 1].data.uncompressed_size
        data: switch compression {
          0 => sized(data_size) decoded(stored) DecodedText (contents text) {
            text: text[remaining] utf8
          }
          8 => sized(data_size) decoded(deflate) DecodedText (contents text) {
            text: text[remaining] utf8
          }
          12 => sized(data_size) decoded(bzip2) DecodedText (contents text) {
            text: text[remaining] utf8
          }
          93 => sized(data_size) decoded(zstd) DecodedText (contents text) {
            text: text[remaining] utf8
          }
          95 => sized(data_size) decoded(xz) DecodedText (contents text) {
            text: text[remaining] utf8
          }
          _ => bytes[data_size]
        }
      }
      0x08074b50 => switch 4294967294 < earlier(body.compressed_size) {
        1 => Zip64DataDescriptor {
          crc32: u32le
              check crc32 over unpacked earlier(body.data) of uncompressed_size bytes when 1 -
                  bit(earlier(body.flags), 0)
          compressed_size: u64le
          uncompressed_size: u64le
        }
        _ => DataDescriptor {
          crc32: u32le
              check crc32 over unpacked earlier(body.data) of uncompressed_size bytes when 1 -
                  bit(earlier(body.flags), 0)
          compressed_size: u32le
          uncompressed_size: u32le
        }
      }
      0x02014b50 => CentralDirectoryFile {
        version_made_by: u16le
        version_needed: u16le
        flags: u16le flags GeneralPurposeFlags {
          bit 0 = encrypted, bit 3 = data descriptor follows, bit 6 = strong encryption,
          bit 11 = UTF-8 names, bit 13 = directory encryption
        }
        compression: u16le enum CompressionMethod {
          0 = stored, 1 = shrunk, 6 = imploded, 8 = deflate, 9 = deflate64, 12 = bzip2,
          14 = lzma, 93 = zstandard, 95 = xz, 98 = ppmd, 99 = aes
        }
        modified_time: u16le  time dos halves(date modified_date, time modified_time) local
        modified_date: u16le  time dos halves(date modified_date, time modified_time) local
        crc32: u32le
        compressed_size: u32le
        uncompressed_size: u32le
        name_length: u16le
        extra_length: u16le
        comment_length: u16le
        disk_number: u16le
        internal_attributes: u16le
        external_attributes: u32le
        local_header_offset: u32le
        name: text[name_length] encoding unknown
        extra: sized(extra_length) repeat(until end) ExtraField (named by id, contents data) {
          id: u16le enum ExtraFieldId {
            0x01 = ZIP64 extended info, 0x07 = AV info, 0x08 = language, 0x09 = OS/2,
            0x0a = NTFS times, 0x0c = OpenVMS, 0x0d = UNIX, 0x0e = stream and fork descriptor,
            0x0f = patch descriptor, 0x14 = PKCS#7 certificates, 0x15 = file signature,
            0x16 = directory signature, 0x17 = strong encryption, 0x19 = certificate list,
            0x65 = IBM attributes, 0x66 = IBM compressed attributes, 0x4690 = POSZIP,
            0x5455 = extended timestamp, 0x5855 = Info-ZIP UNIX, 0x6375 = Unicode comment,
            0x7075 = Unicode name, 0x7855 = Info-ZIP UNIX 2, 0x7875 = Info-ZIP UNIX 3,
            0x9901 = AES encryption, 0xa11e = alignment padding, 0xa220 = growth hint
          }
          size: u16le
          data: sized(size) switch id {
            1 => Zip64Fields {
              uncompressed_size: switch (4294967294 < uncompressed_size) * (7 < remaining) {
                1 => u64le
                _ => bytes[0]
              }
              compressed_size: switch (4294967294 < compressed_size) * (7 < remaining) {
                1 => u64le
                _ => bytes[0]
              }
              local_header_offset: switch (4294967294 < local_header_offset) * (7 < remaining) {
                1 => u64le
                _ => bytes[0]
              }
              disk_number: switch (65534 < disk_number) * (3 < remaining) {
                1 => u32le
                _ => bytes[0]
              }
              rest: bytes[remaining]
            }
            _ => bytes[remaining]
          }
        }
        comment: text[comment_length] encoding unknown
      }
      0x06054b50 => EndOfCentralDirectory {
        disk_number: u16le
        directory_disk: u16le
        entries_on_disk: u16le
        entries_total: u16le
        directory_size: u32le
        directory_offset: u32le
        comment_length: u16le
        comment: text[comment_length] encoding unknown
      }
      0x06064b50 => Zip64End {
        record_size: u64le
        record: sized(record_size) switch 43 < record_size {
          1 => Zip64EndRecord {
            version_made_by: u16le
            version_needed: u16le
            disk_number: u32le
            directory_disk: u32le
            entries_on_disk: u64le
            entries_total: u64le
            directory_size: u64le
            directory_offset: u64le
            extensible_data: bytes[remaining]
          }
          _ => bytes[remaining]
        }
      }
      0x07064b50 => Zip64Locator {
        directory_disk: u32le
        directory_offset: u64le
        disk_count: u32le
      }
      _ => bytes[find("PK")]
    }
  }
}
