type HeapIndirectBlock {
  signature: magic "FHIB"
  version: u8be
  heap_header_address: u64le
  block_offset: bytes[(max_heap_size + 7) / 8]
  children: HeapBlock {
    address: u64le
    row: computed index / table_width
    block_size: switch row {
      0 => computed starting_block_size
      _ => computed starting_block_size << row - 1
    }
    filtered_block: switch (row < direct_rows) * (0 < io_filter_length) {
      1 => FilteredBlock (inline) {
        size: u64le
        filter_mask: u32le
      }
      _ => bytes[0]
    }
    block_rows: switch row < direct_rows {
      1 => computed 0
      _ => computed log2(block_size) - log2(starting_block_size * table_width) + 1
    }
    block: switch row < direct_rows {
      1 => switch 0 < io_filter_length {
        1 => switch address {
          0xffffffffffffffff => bytes[0]
          _ => at(address from origin) bytes[filtered_block.size]
        }
        _ => switch address {
          0xffffffffffffffff => bytes[0]
          _ => at(address from origin) sized(block_size) HeapDirectBlock
        }
      }
      _ => switch address {
        0xffffffffffffffff => bytes[0]
        _ => at(address from origin) HeapIndirectBlock
      }
    }
  }[table_width * block_rows]
  checksum: u32le
}

                      _ => (sized(element_size) bytes[element_size])[run_bytes / element_size]
                    }
                  }
                  _ => (sized(element_size) bytes[element_size])[run_bytes / element_size]
                }
              }
              _ => FilteredChunk (packed hdf5_chunk) {
                bytes: bytes[chunk_size]
              }
            }
          }
          _ => switch child_address {
            0xffffffffffffffff => bytes[0]
            _ => at(child_address from origin) Node
          }
        }
      }
      _ => Entry (named by key_name, contents child) {
        name_offset: u64le
        key_name: at(data_segment_address + name_offset from origin) text until 0x00 utf8
        child_address: u64le
        child: switch child_address {
          0xffffffffffffffff => bytes[0]
          _ => at(child_address from origin) Node
        }
      }
    }[entries_used]
    last_key: switch node_type {
      1 => Key {
        chunk_size: u32le
        filter_mask: u32le
        offsets: u64le[dimensionality]
      }
      _ => Key {
        name_offset: u64le
      }
    }
  }
  'SNOD' => SymbolTableNode {
    signature: magic "SNOD"
    version: u8be
    reserved: u8be
    symbol_count: u16le
    symbols: Link (named by name, contents object) {
      link_name_offset: u64le
      object_header_address: u64le
      cache_type: u32le enum CacheType {0 = nothing cached, 1 = group, 2 = symbolic link}
      reserved: u32le
      scratch: switch cache_type {
        1 => GroupCache (inline) {
          cached_btree_address: u64le
          cached_heap_address: u64le
        }
        2 => LinkCache {
          link_value_offset: u32le
          link_value: at(data_segment_address + link_value_offset from origin) text until 0x00
              utf8
          unused: bytes[12]
        }
        _ => bytes[16]
      }
      name: at(data_segment_address + link_name_offset from origin) text until 0x00 utf8
      object: switch object_header_address {
        0xffffffffffffffff => bytes[0]
        _ => at(object_header_address from origin) ObjectHeader
      }
    }[symbol_count]
  }
  _ => UnknownNode {
    signature: text[4] utf8
  }
}

type ObjectHeader = switch peek(u8le) {
  1 => ObjectHeader {
    version: u8be
    reserved: u8be
    message_count: u16le
    reference_count: u32le
    header_size: u32le
    padding: bytes[4]
    messages: sized(header_size) repeat(until end) Message
  }
  79 => ObjectHeader {
    signature: magic "OHDR"
    version: u8be
    header_flags: u8be flags HeaderFlags {
      bit 2 = attribute creation order tracked, bit 3 = attribute creation order indexed,
      bit 4 = non-default attribute storage, bit 5 = times stored
    }
    times: switch header_flags / 32 - header_flags / 64 * 2 {
      1 => Times (inline) {
        access_time: u32le  time unix utc
        modification_time: u32le  time unix utc
        change_time: u32le  time unix utc
        birth_time: u32le  time unix utc
      }
      _ => bytes[0]
    }
    attribute_storage: switch header_flags / 16 - header_flags / 32 * 2 {
      1 => AttributeStorage (inline) {
        max_compact: u16le
        min_dense: u16le
      }
      _ => bytes[0]
    }
    chunk_size: switch header_flags - header_flags / 4 * 4 {
      0 => u8le
      1 => u16le
      2 => u32le
      3 => u64le
      _ => bytes[0]
    }
    messages: sized(chunk_size) repeat(until end) MessageV2
    checksum: u32le
  }
  _ => UnknownObjectHeader {
    version: u8be
  }
}
