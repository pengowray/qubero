type AudioSampleEntry {
  // QuickTime sound description version: 0, 1 or 2. Versions 1 and 2 add fields after
      sample_rate. ISO 14496-12 calls this entry_version.
  version: u16be
  revision_level: u16be
  // QuickTime: who made the codec. ISO 14496-12 reserves these four bytes and requires zero.
  vendor: text[4] latin1
  channel_count: u16be
  // Bits per sample per channel, for uncompressed audio. 16 for anything compressed.
  sample_size: u16be
  // QuickTime: -2 in a version 2 entry, 0 for uncompressed, -1 for a compressed version 1
      entry.
  compression_id: i16be
  packet_size: u16be
  // Samples per second. In a version 2 entry this is 1.0, and the rate is in audio_sample_rate.
  sample_rate: ufixed 16.16be
  v1: optional(when version == 1) SoundDescriptionV1 {
    samples_per_packet: u32be
    bytes_per_packet: u32be
    bytes_per_frame: u32be
    bytes_per_sample: u32be
  }
  v2: optional(when version == 2) SoundDescriptionV2 {
    size_of_struct_only: u32be
    audio_sample_rate: f64be
    audio_channels: u32be
    always_7f000000: u32be
    const_bits_per_channel: u32be
    format_specific_flags: u32be flags LpcmFlags {
      bit 0 = float, bit 1 = big-endian, bit 2 = signed integer, bit 3 = packed,
      bit 4 = aligned high, bit 5 = non-interleaved
    }
    const_bytes_per_audio_packet: u32be
    const_lpcm_frames_per_audio_packet: u32be
  }
  boxes: repeat(until end) Box
}

        0x61766331 = avc1 (H.264), 0x61766333 = avc3 (H.264), 0x68766331 = hvc1 (HEVC),
        0x68657631 = hev1 (HEVC), 0x64766831 = dvh1 (Dolby Vision HEVC),

        0x76703038 = vp08 (VP8), 0x76703039 = vp09 (VP9), 0x6d703476 = mp4v (MPEG-4 Visual),

        0x616c6163 = alac (Apple Lossless), 0x4f707573 = Opus, 0x664c6143 = fLaC (FLAC),
        0x61632d33 = ac-3 (Dolby Digital), 0x65632d33 = ec-3 (Dolby Digital Plus),

type H264Sample = switch nal_length_size == 1 or nal_length_size == 2 or nal_length_size == 4 {
  1 => repeat(until end) LengthPrefixedNalUnit (named by nal) {
    // The size in bytes of the NAL unit that follows.
    length: u(nal_length_size * 8)be

  }
  _ => bytes[remaining]
}

type HevcNalUnit (named by unit_type) {
  forbidden_zero: u1be
  unit_type: u6be enum HevcNalType {
    0 = TRAIL_N slice, 1 = TRAIL_R slice, 2 = TSA_N slice, 3 = TSA_R slice, 4 = STSA_N slice,
    5 = STSA_R slice, 6 = RADL_N slice, 7 = RADL_R slice, 8 = RASL_N slice, 9 = RASL_R slice,
    16 = BLA_W_LP slice, 17 = BLA_W_RADL slice, 18 = BLA_N_LP slice, 19 = IDR_W_RADL slice,
    20 = IDR_N_LP slice, 21 = CRA slice, 32 = VPS, 33 = SPS, 34 = PPS,
    35 = access unit delimiter, 36 = end of sequence, 37 = end of bitstream, 38 = filler,
    39 = prefix SEI, 40 = suffix SEI
  }
  layer_id: u6be
  temporal_id_plus1: u3be
  rbsp: switch unit_type == 39 or unit_type == 40 {
    1 => SeiRbsp {
      messages: repeat(while remaining > 1) SeiMessage
          (named by payload_type, machinery type_code size_code) {
        type_code: SeiPayloadType (named by value) {
          ff_bytes: repeat(while peek(u8be) == 255) u8be
          last_byte: u8be
          value: computed count of ff_bytes * 255 + last_byte
        }
        payload_type: computed type_code.value enum SeiType {
          0 = buffering period, 1 = picture timing, 2 = pan-scan rectangle, 3 = filler payload,
          4 = user data registered (ITU-T T.35), 5 = user data unregistered,
          6 = recovery point, 19 = film grain characteristics, 45 = frame packing arrangement,
          47 = display orientation, 128 = structure of pictures, 129 = active parameter sets,
          132 = decoded picture hash, 137 = mastering display colour volume,
          144 = content light level, 147 = alternative transfer characteristics
        }
        size_code: SeiPayloadSize (named by value) {
          ff_bytes: repeat(while peek(u8be) == 255) u8be
          last_byte: u8be
          value: computed count of ff_bytes * 255 + last_byte
        }
        payload: sized(min(size_code.value, remaining)) switch type_code.value {
          5 => UnregisteredUserData {
            // A UUID for whoever wrote this SEI message. x264 writes its version and settings
                as text after the UUID.
            uuid: bytes[16]
            text: text[remaining] padded with 0x00 utf8
          }
          _ => bytes[remaining]
        }
      }
      // The stop bit and the zero bits after it, 0x80.
      trailing_bits: u8be
    }
    _ => bytes[remaining]
  }
}

type SampleContent = switch format {
  'avc1' => H264Sample
  'avc2' => H264Sample
  'avc3' => H264Sample
  'avc4' => H264Sample
  'hvc1' => HevcSample
  'hev1' => HevcSample

  'dvhe' => HevcSample
  'av01' => repeat(until end) Obu (named by obu_type) {
    forbidden_bit: u1be
    obu_type: u4be enum ObuType {
      1 = sequence header, 2 = temporal delimiter, 3 = frame header, 4 = tile group,
      5 = metadata, 6 = frame, 7 = redundant frame header, 8 = tile list, 15 = padding
    }
    extension_flag: u1be
    has_size_field: u1be
    reserved_1bit: u1be

    spatial_id: optional(when extension_flag) u2be

type VisualSampleEntry {
  version: u16be
  revision_level: u16be
  // QuickTime: who made the codec, such as appl or FFMP. Zero in an MP4.
  vendor: text[4] latin1
  temporal_quality: u32be
  spatial_quality: u32be
  width: u16be
  height: u16be
  // Pixels per inch, 72 by convention.
  horiz_resolution: ufixed 16.16be
  vert_resolution: ufixed 16.16be
  // Reserved, zero.
  data_size: u32be
  // How many frames each sample holds, 1 for every common codec.
  frame_count: u16be
  compressor_name: CompressorName (named by text) {
    length: u8be
    text: text[31] padded with 0x00 utf8
  }
  // Bits per pixel: 24 for colour with no alpha.
  depth: u16be
  // -1: no colour table.
  color_table_id: i16be
  boxes: repeat(until end) Box
}
